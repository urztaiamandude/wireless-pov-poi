ESP32-S3 Gateway (Simplified Example)

Educational example showing basic WiFi AP and UART bridge architecture.

#include <WiFi> #include <WebServer> #include <SPIFFS> #include <ArduinoJson> 
// ==========================================
// SIMPLIFIED EXAMPLE - NOT PRODUCTION CODE
// For production, see: esp32_firmware/esp32_firmware.ino
// ==========================================

#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <SPIFFS.h>
#include <ArduinoJson.h>

const char* ssid = "POV-POI-WiFi";
const char* password = "povpoi123";

#define TEENSY_SERIAL Serial2
#define SERIAL_BAUD 115200

WebServer server(80);

void sendTeensyCommand(uint8_t cmd, const uint8_t* data, size_t len) {
  TEENSY_SERIAL.write(0xFF);  // Start marker
  TEENSY_SERIAL.write(cmd);   // Command byte
  TEENSY_SERIAL.write(len);   // Data length
  TEENSY_SERIAL.write(data, len);
  TEENSY_SERIAL.write(0xFE);  // End marker
}

void setup() {
  Serial.begin(115200);
  TEENSY_SERIAL.begin(SERIAL_BAUD, SERIAL_8N1, 44, 43);
  
  if(!SPIFFS.begin(true)) {
    Serial.println("SPIFFS Mount Failed");
  }

  WiFi.softAP(ssid, password);
  Serial.print("AP IP: ");
  Serial.println(WiFi.softAPIP());

  server.on("/api/status", HTTP_GET, [](){
    server.send(200, "application/json", 
      "{\"mode\":1,\"brightness\":128,\"connected\":true}");
  });

  server.on("/api/brightness", HTTP_POST, [](){
    if (server.hasArg("plain")) {
      String body = server.arg("plain");
      TEENSY_SERIAL.println(body);
      server.send(200, "application/json", "{\"status\":\"ok\"}");
    }
  });

  server.serveStatic("/", SPIFFS, "/").setDefaultFile("index.html");
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  server.handleClient();
}

Teensy 4.1 POV Engine (Simplified Example)

Educational example showing basic LED control and serial communication.

#include <FastLED> #include <SdFat> #include <ArduinoJson> 
// ==========================================
// SIMPLIFIED EXAMPLE - NOT PRODUCTION CODE
// For production, see: teensy_firmware/teensy_firmware.ino
// ==========================================

#include <Arduino.h>
#include <FastLED.h>
#include <SdFat.h>
#include <ArduinoJson.h>

#define NUM_LEDS 32
#define DATA_PIN 11
#define CLOCK_PIN 13
#define SERIAL_BAUD 115200
#define DEFAULT_BRIGHTNESS 128

CRGB leds[NUM_LEDS];
SdFs sd;

uint8_t currentBrightness = DEFAULT_BRIGHTNESS;
uint8_t currentMode = 0;

void displayPattern(uint8_t patternIndex) {
  static uint8_t hue = 0;
  
  // Rainbow pattern across all display LEDs
  for (int i = 0; i < NUM_LEDS; i++) {
    leds[i] = CHSV(hue + (i * 10), 255, 255);
  }
  hue++;
  FastLED.show();
}

void clearLEDs() {
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();
}

void setup() {
  Serial1.begin(SERIAL_BAUD);  // ESP32 communication
  Serial.begin(115200);        // Debug

  FastLED.addLeds<APA102, DATA_PIN, CLOCK_PIN, BGR>(leds, NUM_LEDS);
  FastLED.setBrightness(currentBrightness);
  clearLEDs();

  if (!sd.begin(SdioConfig(FIFO_SDIO))) {
    Serial.println("SD card initialization failed");
  }

  Serial.println("Teensy POV Engine Ready");
}

void loop() {
  if (Serial1.available()) {
    String line = Serial1.readStringUntil('\n');
    StaticJsonDocument<256> doc;
    
    if (deserializeJson(doc, line) == DeserializationError::Ok) {
      String cmd = doc["cmd"] | "";
      if (cmd == "brightness") {
        currentBrightness = doc["value"] | 128;
        FastLED.setBrightness(currentBrightness);
      } else if (cmd == "mode") {
        currentMode = doc["value"] | 0;
      }
    }
  }
  
  if (currentMode == 2) {
    displayPattern(0);
    delay(16);
  }
}

⚠️ CRITICAL SAFETY

High-density APA102 LED strips can pull significant amperage. Do not power the LEDs directly from the Teensy 5V pin. Use a dedicated 5V power supply and share the ground with both boards. For portable POV systems, use 18650 cells with a boost converter.

OTA Feature Not Yet Implemented

The firmware manager UI is ready, but the backend OTA endpoints (/api/ota/esp32 and /api/ota/teensy) are not yet implemented in the current ESP32 firmware. To enable wireless updates, these endpoints need to be added to esp32_firmware/esp32_firmware.ino.

For now, use USB cable and Arduino IDE/PlatformIO to update firmware manually.
